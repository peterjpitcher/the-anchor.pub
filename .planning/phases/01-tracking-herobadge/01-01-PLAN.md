```tsx
import { Badge } from '@/components/ui/primitives/Badge'
import { cn } from '@/lib/utils'
import ssot from '@/SSOT.json'

// Pull badge content from SSOT.json at build time (per D-03)
const GOOGLE_RATING = ssot.ratings.google.rating
const GOOGLE_REVIEW_COUNT = ssot.ratings.google.review_count

interface HeroBadgeProps {
  className?: string
}

/**
 * HeroBadge — displays the standard set of trust badges (Google rating, review count).
 * All pages show the same badges (per D-02). Content comes from SSOT.json (per D-03).
 * Wraps the Badge primitive (per D-01).
 */
export function HeroBadge({ className }: HeroBadgeProps) {
  return (
    <div className={cn('flex flex-wrap items-center justify-center gap-2', className)}>
      <Badge variant="primary" size="sm">
        {GOOGLE_RATING}/5 on Google
      </Badge>
      <Badge variant="outline" size="sm">
        {GOOGLE_REVIEW_COUNT} reviews
      </Badge>
    </div>
  )
}

// Preserve HeroItem export for backward compatibility with ManagersSpecial and MenuRenderer.
// HeroItem is used for menu item badges (NEW, Featured, etc.) and is unrelated to the
// trust-badge HeroBadge above. Keep it functional but route through Badge primitive.
interface HeroItemProps {
  children: React.ReactNode
  badgeText?: string
  badgeVariant?: 'new' | 'featured' | 'special' | 'limited'
  showBadge?: boolean
  className?: string
}

const itemBadgeVariantMap: Record<string, 'error' | 'warning' | 'success' | 'primary'> = {
  new: 'error',
  featured: 'warning',
  special: 'success',
  limited: 'primary'
}

export function HeroItem({
  children,
  badgeText = 'NEW',
  badgeVariant = 'new',
  showBadge = false,
  className = ''
}: HeroItemProps) {
  if (!showBadge) {
    return <>{children}</>
  }

  return (
    <div className={cn('relative', className)}>
      <Badge
        variant={itemBadgeVariantMap[badgeVariant] || 'error'}
        size="sm"
        className="absolute -top-2 -left-2 z-10 transform -rotate-12 shadow-md uppercase hidden md:inline-flex"
      >
        {badgeText}
      </Badge>
      {children}
    </div>
  )
}
```

Key design choices:
- HeroBadge is now a Server Component (no 'use client') since it has no interactivity.
- Badge variant mapping: `primary` for the gold-tinted rating badge, `outline` for the subtle review count.
- HeroItem preserved for backward compat with ManagersSpecial.tsx and MenuRenderer.tsx. Variant mapping: new=error (red tint), featured=warning (yellow), special=success (gold-green), limited=primary (gold).
- The old `position` prop and responsive absolute/inline logic is removed. The new HeroBadge is always a flex row of badges.
